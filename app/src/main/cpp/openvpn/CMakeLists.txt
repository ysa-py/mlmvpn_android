cmake_minimum_required(VERSION 3.20)
project(mlmopenvpn LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(DEPS "${CMAKE_CURRENT_LIST_DIR}/../../../../../third_party/openvpn3")
file(READ "${DEPS}/source-lock.json" SOURCE_LOCK)
string(JSON CORE_COMMIT GET "${SOURCE_LOCK}" coreCommit)
set(OPENVPN_CORE_DIR "${DEPS}/openvpn3-${CORE_COMMIT}" CACHE PATH "Pinned OpenVPN source")
set(CORE "${OPENVPN_CORE_DIR}")
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(GEN_FILES OFF CACHE BOOL "" FORCE)
set(MBEDTLS_FATAL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/mbedtls-mbedtls-3.6.7" mbedtls EXCLUDE_FROM_ALL)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${DEPS}/fmt-11.1.4" fmt EXCLUDE_FROM_ALL)
add_library(mlmopenvpn SHARED bridge.cpp "${CORE}/client/ovpncli.cpp"
    "${CORE}/openvpn/crypto/data_epoch.cpp" "${DEPS}/lz4-1.10.0/lib/lz4.c")
target_include_directories(mlmopenvpn PRIVATE "${CORE}" "${DEPS}/asio-asio-1-30-2/asio/include"
    "${DEPS}/lz4-1.10.0/lib")
target_compile_definitions(mlmopenvpn PRIVATE USE_ASIO ASIO_STANDALONE
    USE_MBEDTLS HAVE_LZ4 OPENVPN_CORE_API_VISIBILITY_HIDDEN)
target_compile_options(mlmopenvpn PRIVATE -O2 -ffunction-sections -fdata-sections)
target_link_options(mlmopenvpn PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
target_link_libraries(mlmopenvpn PRIVATE mbedtls mbedx509 mbedcrypto fmt::fmt log)

option(MLM_PROFILE_CHECK "Build the offline device profile check" OFF)
if(MLM_PROFILE_CHECK)
    add_executable(mlmopenvpn-check profile_check.cpp "${CORE}/client/ovpncli.cpp"
        "${CORE}/openvpn/crypto/data_epoch.cpp" "${DEPS}/lz4-1.10.0/lib/lz4.c")
    target_include_directories(mlmopenvpn-check PRIVATE "${CORE}" "${DEPS}/asio-asio-1-30-2/asio/include" "${DEPS}/lz4-1.10.0/lib")
    target_compile_definitions(mlmopenvpn-check PRIVATE USE_ASIO ASIO_STANDALONE USE_MBEDTLS HAVE_LZ4)
    target_link_libraries(mlmopenvpn-check PRIVATE mbedtls mbedx509 mbedcrypto fmt::fmt log)
endif()
